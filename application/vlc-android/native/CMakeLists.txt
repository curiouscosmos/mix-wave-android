cmake_minimum_required(VERSION 3.18)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")

    target_link_options(caption_seek_probe PRIVATE -Wl,-z,max-page-size=16384)
    target_link_libraries(caption_seek_probe PRIVATE dl)
endif()
